/// ---------------------------------------------------------------------------
/// FLUTTER MATERIAL PAGE ROUTE GENERATOR
/// ---------------------------------------------------------------------------
class AppRouteGenerator {
  static Route<dynamic> generateRoute(RouteSettings settings) {
    final Uri uri = Uri.parse(settings.name ?? AppRoutes.login);
    final String path = uri.path;

    switch (path) {
      // Phase 1: Authentication & Device Governance
      case AppRoutes.login:
        return _buildRoute(const UnifiedLoginScreen(), settings);
      case AppRoutes.twoFactorOtp:
        return _buildRoute(const TwoFactorOtpScreen(), settings);
      case AppRoutes.securityLockout:
        return _buildRoute(const SecurityLockoutScreen(), settings);
      case AppRoutes.passwordReset:
        return _buildRoute(const PasswordResetScreen(), settings);
      case AppRoutes.deviceManagement:
        return _buildRoute(const DeviceManagementScreen(), settings);

      // Phase 2: Parent & Student Portals
      case AppRoutes.parentHome:
        return _buildRoute(const ParentHomeScreen(), settings);
      case AppRoutes.feeLedger:
        return _buildRoute(const FeeLedgerScreen(), settings);
      case AppRoutes.attendanceMatrix:
        return _buildRoute(const AttendanceMatrixScreen(), settings);
      case AppRoutes.studentLeave:
        return _buildRoute(const StudentLeaveScreen(), settings);
      case AppRoutes.noticeBoard:
        return _buildRoute(const NoticeBoardScreen(), settings);
      case AppRoutes.academicCalendar:
        return _buildRoute(const AcademicCalendarScreen(), settings);
      case AppRoutes.notificationCenter:
        return _buildRoute(const NotificationCenterScreen(), settings);
      case AppRoutes.libraryCirculation:
        return _buildRoute(const LibraryCirculationScreen(), settings);
      case AppRoutes.busTransit:
        return _buildRoute(const BusTransitScreen(), settings);

      // Phase 3: Faculty Operations
      case AppRoutes.teacherDashboard:
        return _buildRoute(const TeacherDashboardScreen(), settings);
      case AppRoutes.dailyRollCall:
        return _buildRoute(const DailyRollCallScreen(), settings);
      case AppRoutes.marksEntryDesk:
        return _buildRoute(const MarksEntryDeskScreen(), settings);
      case AppRoutes.facultyTimetable:
        return _buildRoute(const FacultyTimetableScreen(), settings);
      case AppRoutes.facultyLeave:
        return _buildRoute(const FacultyLeaveScreen(), settings);
      case AppRoutes.announcementAuthoring:
        return _buildRoute(const AnnouncementAuthoringScreen(), settings);

      // Phase 4: Executive Leadership & Self-Service
      case AppRoutes.principalCommand:
        return _buildRoute(const PrincipalCommandScreen(), settings);
      case AppRoutes.announcementApproval:
        return _buildRoute(const AnnouncementApprovalScreen(), settings);
      case AppRoutes.unifiedSearch:
        return _buildRoute(const UnifiedSearchScreen(), settings);
      case AppRoutes.staffDirectory:
        return _buildRoute(const StaffDirectoryScreen(), settings);
      case AppRoutes.studentSelfService:
        return _buildRoute(const StudentSelfServiceScreen(), settings);
      case AppRoutes.studentIdCard:
        return _buildRoute(const StudentIdCardScreen(), settings);
      case AppRoutes.admissionsDesk:
        return _buildRoute(const AdmissionsDeskScreen(), settings);

      // Dynamic Parameterized Routes
      default:
        // Match /student/:studentId/dossier
        if (path.startsWith('/student/') && path.endsWith('/dossier')) {
          final studentId = path.split('/')[2];
          return _buildRoute(StudentDossierScreen(studentId: studentId), settings);
        }
        // Match /student/:studentId/report-card
        if (path.startsWith('/student/') && path.endsWith('/report-card')) {
          final studentId = path.split('/')[2];
          return _buildRoute(AcademicReportScreen(studentId: studentId), settings);
        }
        // Match /student/:studentId/timetable
        if (path.startsWith('/student/') && path.endsWith('/timetable')) {
          final studentId = path.split('/')[2];
          return _buildRoute(ClassTimetableScreen(studentId: studentId), settings);
        }
        // Match /fees/receipt/:receiptId
        if (path.startsWith('/fees/receipt/')) {
          final receiptId = path.split('/').last;
          return _buildRoute(FeeReceiptScreen(receiptId: receiptId), settings);
        }

        return _buildRoute(const NotFoundScreen(), settings);
    }
  }

  static MaterialPageRoute _buildRoute(Widget screen, RouteSettings settings) {
    return MaterialPageRoute(
      settings: settings,
      builder: (context) => screen,
    );
  }
}

/// ---------------------------------------------------------------------------
/// DYNAMIC POST-AUTHENTICATION ROLE RESOLVER
/// Evaluates backend relations from /accounts/login/ response and deep-links:
/// ---------------------------------------------------------------------------
class RoleBasedNavigator {
  static void routeUser(BuildContext context, {
    required bool isSuperuser,
    required List<String> userRoles,
  }) {
    if (isSuperuser || userRoles.contains('PRINCIPAL')) {
      Navigator.pushReplacementNamed(context, AppRoutes.principalCommand);
    } else if (userRoles.contains('CLASS_TEACHER') || userRoles.contains('TEACHER')) {
      Navigator.pushReplacementNamed(context, AppRoutes.teacherDashboard);
    } else if (userRoles.contains('PARENT')) {
      Navigator.pushReplacementNamed(context, AppRoutes.parentHome);
    } else if (userRoles.contains('STUDENT')) {
      Navigator.pushReplacementNamed(context, AppRoutes.studentSelfService);
    } else if (userRoles.contains('STAFF') || userRoles.contains('ADMISSIONS')) {
      Navigator.pushReplacementNamed(context, AppRoutes.admissionsDesk);
    } else {
      Navigator.pushReplacementNamed(context, AppRoutes.noticeBoard);
    }
  }
}

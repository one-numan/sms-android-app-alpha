// 2. USER ROLES & AUTHENTICATION STATE MODEL (Django 5.1.4 Aligned)
// -----------------------------------------------------------------------------
enum AuthStatus { unauthenticated, pending2fa, lockedOut, authenticated }

enum UserRole { parent, teacher, principal, student, staff, superuser }

class AuthStateNotifier extends ChangeNotifier {
  AuthStatus _status = AuthStatus.unauthenticated;
  UserRole? _primaryRole;
  String? _username;
  int? _lockoutRemainingSeconds;

  AuthStatus get status => _status;
  UserRole? get primaryRole => _primaryRole;
  String? get username => _username;
  int? get lockoutRemainingSeconds => _lockoutRemainingSeconds;

  bool get isAuthenticated => _status == AuthStatus.authenticated;
  bool get isPending2FA => _status == AuthStatus.pending2fa;
  bool get isLockedOut => _status == AuthStatus.lockedOut;

  void authenticate({required UserRole role, required String username}) {
    _status = AuthStatus.authenticated;
    _primaryRole = role;
    _username = username;
    notifyListeners();
  }

  void requireTwoFactorOtp({required String username}) {
    _status = AuthStatus.pending2fa;
    _username = username;
    notifyListeners();
  }

  void enforceSecurityLockout({required int remainingSeconds}) {
    _status = AuthStatus.lockedOut;
    _lockoutRemainingSeconds = remainingSeconds;
    notifyListeners();
  }

  void signOut() {
    _status = AuthStatus.unauthenticated;
    _primaryRole = null;
    _username = null;
    _lockoutRemainingSeconds = null;
    notifyListeners();
  }
}

// 3. PRODUCTION GO_ROUTER CONFIGURATION & ROLE GUARDS
// -----------------------------------------------------------------------------
final GlobalKey<NavigatorState> rootNavigatorKey = GlobalKey<NavigatorState>(debugLabel: 'root');
final GlobalKey<NavigatorState> shellNavigatorKey = GlobalKey<NavigatorState>(debugLabel: 'shell');

GoRouter createOnpsRouter(AuthStateNotifier authState) {
  return GoRouter(
    navigatorKey: rootNavigatorKey,
    initialLocation: AppRoutePaths.login,
    refreshListenable: authState,
    debugLogDiagnostics: true,
    
    // Top-Level Redirect Engine (Single-Door Authentication & Role Guarding)
    redirect: (BuildContext context, GoRouterState state) {
      final matchedLocation = state.matchedLocation;
      final authStatus = authState.status;

      // 1. Handle Security Lockout (Screen 02b)
      if (authStatus == AuthStatus.lockedOut) {
        if (matchedLocation != AppRoutePaths.securityLockout) {
          return AppRoutePaths.securityLockout;
        }
        return null;
      }

      // 2. Handle 2-Factor OTP Challenge (Screen 02)
      if (authStatus == AuthStatus.pending2fa) {
        if (matchedLocation != AppRoutePaths.twoFactorOtp) {
          return AppRoutePaths.twoFactorOtp;
        }
        return null;
      }

      // 3. Handle Unauthenticated Requests
      final isPublicAuthRoute = matchedLocation == AppRoutePaths.login ||
          matchedLocation == AppRoutePaths.passwordReset;

      if (!authState.isAuthenticated) {
        return isPublicAuthRoute ? null : AppRoutePaths.login;
      }

      // 4. Authenticated users attempting to visit login/password-reset get redirected
      if (isPublicAuthRoute) {
        switch (authState.primaryRole) {
          case UserRole.principal:
          case UserRole.superuser:
            return AppRoutePaths.principalCommand;
          case UserRole.teacher:
            return AppRoutePaths.teacherDashboard;
          case UserRole.parent:
            return AppRoutePaths.parentHome;
          case UserRole.student:
            return AppRoutePaths.studentSelfService;
          case UserRole.staff:
            return AppRoutePaths.admissionsDesk;
          default:
            return AppRoutePaths.noticeBoard;
        }
      }

      // 5. Role-based Route Protection Guard
      final role = authState.primaryRole;
      if (matchedLocation.startsWith('/principal') &&
          role != UserRole.principal &&
          role != UserRole.superuser) {
        return AppRoutePaths.parentHome; // Unauthorized bounce
      }

      if (matchedLocation.startsWith('/teacher') &&
          role != UserRole.teacher &&
          role != UserRole.principal &&
          role != UserRole.superuser) {
        return AppRoutePaths.parentHome; // Unauthorized bounce
      }

      return null;
    },

    // Route Definitions
    routes: <RouteBase>[
      // --- Phase 1: Authentication & Device Security ---
      GoRoute(
        path: AppRoutePaths.login,
        name: 'login',
        builder: (context, state) => const PlaceholderScreen(title: '01 - Unified Login Gateway'),
      ),
      GoRoute(
        path: AppRoutePaths.twoFactorOtp,
        name: 'twoFactorOtp',
        builder: (context, state) => const PlaceholderScreen(title: '02 - 2-Factor OTP Verification'),
      ),
      GoRoute(
        path: AppRoutePaths.securityLockout,
        name: 'securityLockout',
        builder: (context, state) => const PlaceholderScreen(title: '02b - Security Lockout & Cooldown'),
      ),
      GoRoute(
        path: AppRoutePaths.passwordReset,
        name: 'passwordReset',
        builder: (context, state) => const PlaceholderScreen(title: '03 - Password Reset & Recovery'),
      ),
      GoRoute(
        path: AppRoutePaths.deviceManagement,
        name: 'deviceManagement',
        builder: (context, state) => const PlaceholderScreen(title: '04 - Multi-Device Management'),
      ),

      // --- Phase 2: Student & Parent Scholastic Portals ---
      GoRoute(
        path: AppRoutePaths.parentHome,
        name: 'parentHome',
        builder: (context, state) => const PlaceholderScreen(title: '05 - Parent Portal Home'),
      ),
      GoRoute(
        path: AppRoutePaths.studentDossier,
        name: 'studentDossier',
        builder: (context, state) {
          final studentId = state.pathParameters['studentId'] ?? 'ADM-2024-0412';
          return PlaceholderScreen(title: '06 - Student Dossier (ID: $studentId)');
        },
      ),
      GoRoute(
        path: AppRoutePaths.academicReport,
        name: 'academicReport',
        builder: (context, state) {
          final studentId = state.pathParameters['studentId'] ?? 'ADM-2024-0412';
          return PlaceholderScreen(title: '07 - Terminal Report Card (ID: $studentId)');
        },
      ),
      GoRoute(
        path: AppRoutePaths.classTimetable,
        name: 'classTimetable',
        builder: (context, state) {
          final studentId = state.pathParameters['studentId'] ?? 'ADM-2024-0412';
          return PlaceholderScreen(title: '08 - Class Timetable (ID: $studentId)');
        },
      ),
      GoRoute(
        path: AppRoutePaths.feeLedger,
        name: 'feeLedger',
        builder: (context, state) => const PlaceholderScreen(title: '09 - Fee Ledger & UPI Payment'),
      ),
